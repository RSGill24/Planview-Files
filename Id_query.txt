SELECT 
    [Strategy Seq ID],
    LEN([Strategy Seq ID])                          AS [Len],
    DATALENGTH([Strategy Seq ID])                   AS [DataLen],
    UNICODE(LEFT([Strategy Seq ID], 1))             AS [FirstCharUnicode],
    UNICODE(RIGHT([Strategy Seq ID], 1))            AS [LastCharUnicode]
FROM [PlanviewMigration_Output].[Initiatives_v12]
WHERE DATALENGTH([Strategy Seq ID]) > LEN([Strategy Seq ID]) * 2;
